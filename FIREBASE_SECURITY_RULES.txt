// Firebase Firestore Security Rules
// Copy and paste this into your Firebase Console -> Firestore Database -> Rules tab

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // Allow users to read and write their own user document and all nested subcollections
    // (This recursively covers: /users/{userId}, /users/{userId}/addresses, /users/{userId}/cart,
    //  /users/{userId}/orders, /users/{userId}/paymentMethods, /users/{userId}/notifications)
    match /users/{userId}/{document=**} {
      allow read, write: if request.auth.uid == userId;
    }
    
    // Allow public read for products
    match /products/{document=**} {
      allow read: if true;
      allow write: if false; // Only backend can write
    }
  }
}
